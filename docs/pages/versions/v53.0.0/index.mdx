import React, { useState } from 'react';
import { View, Text, ImageBackground, StyleSheet, Image, TouchableOpacity, Share, ScrollView, Dimensions } from 'react-native';
import { Audio } from 'expo-av';

const { width } = Dimensions.get('window');

const wallpapers = [
  require('./assets/players/bellingham.jpg'),
  require('./assets/players/vinicius.jpg'),
  require('./assets/players/rodrygo.jpg'),
  require('./assets/players/courtois.jpg'),
];

export default function App() {
  const [sound, setSound] = useState();

  const playRingtone = async () => {
    const { sound } = await Audio.Sound.createAsync(
      require('./assets/ringtones/hala_madrid.mp3')
    );
    setSound(sound);
    await sound.playAsync();
  };

  const shareApp = async () => {
    await Share.share({
      message: 'Download Madrid Spirit 2025 - Real Madrid Theme App! Created by Fresh7amanie ⚪️⚽️',
    });
  };

  return (
    <ScrollView style={styles.container}>
      <Text style={styles.title}>Madrid Spirit 2025</Text>
      {wallpapers.map((wallpaper, index) => (
        <ImageBackground key={index} source={wallpaper} style={styles.wallpaper}>
          <Text style={styles.playerTag}>HALA MADRID!</Text>
        </ImageBackground>
      ))}
      <View style={styles.buttonRow}>
        <TouchableOpacity style={styles.button} onPress={playRingtone}>
          <Text style={styles.buttonText}>Play Ringtone</Text>
        </TouchableOpacity>
        <TouchableOpacity style={styles.button} onPress={shareApp}>
          <Text style={styles.buttonText}>Share App</Text>
        </TouchableOpacity>
      </View>
      <Text style={styles.signature}>Created by Fresh7amanie</Text>
    </ScrollView>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, backgroundColor: '#000' },
  title: {
    fontSize: 24,
    color: '#fff',
    textAlign: 'center',
    marginVertical: 20,
    fontWeight: 'bold',
  },
  wallpaper: {
    width: width,
    height: 300,
    justifyContent: 'flex-end',
    alignItems: 'center',
    marginBottom: 20,
  },
  playerTag: {
    backgroundColor: 'rgba(255,255,255,0.7)',
    padding: 8,
    fontSize: 18,
    fontWeight: 'bold',
    marginBottom: 10,
  },
  buttonRow: {
    flexDirection: 'row',
    justifyContent: 'space-around',
    marginVertical: 20,
  },
  button: {
    backgroundColor: '#fff',
    padding: 12,
    borderRadius: 8,
  },
  buttonText: {
    color: '#000',
    fontWeight: 'bold',
  },
  signature: {
    color: '#aaa',
    fontSize: 12,
    textAlign: 'center',
    marginBottom: 20,
  },
});
